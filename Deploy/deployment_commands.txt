# ITC5205 Deployment Commands
# Run these on the EC2 Ubuntu build server from ~/lambda-yolov5.

# Build the container image
docker build --platform linux/amd64 -t itc5205-yolov5-lambda:latest .

# Confirm local image
docker images

# Set variables
ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text)
REGION=us-east-1
REPO=itc5205-yolov5-lambda

# Authenticate Docker with ECR
aws ecr get-login-password --region $REGION | docker login --username AWS --password-stdin $ACCOUNT_ID.dkr.ecr.$REGION.amazonaws.com

# Tag and push image
docker tag itc5205-yolov5-lambda:latest $ACCOUNT_ID.dkr.ecr.$REGION.amazonaws.com/$REPO:latest
docker push $ACCOUNT_ID.dkr.ecr.$REGION.amazonaws.com/$REPO:latest

# After push, open AWS Lambda and deploy the new image from ECR.
